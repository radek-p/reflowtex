% Preamble for the \mypopover example (content/docs/typesetting/footnotes.md).
\usepackage{microtype}
\usepackage{reflowtex}
% \mypopover{label}{note}: a note behind a button in the text. The button is
% a widget; its label and the note are asides, typeset by TeX out of the
% flow, and found by the widget through for=… (examples/popover.js). In
% print, a footnote.
\newcounter{mypopover}
\NewDocumentCommand\mypopover{m +m}{%
  \ifreflowtex
    \stepcounter{mypopover}%
    \webwidget{popover:\themypopover}%
    \webaside[for=\themypopover]{popover-label}{#1}%
    \webaside[for=\themypopover]{popover-note}{\footnotesize #2}%
  \else
    \footnote{#2}%
  \fi}
