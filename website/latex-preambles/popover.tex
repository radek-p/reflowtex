% Preamble for the \mypopover example (content/docs/web-first/inline-buttons.md).
\usepackage{microtype}
\usepackage{reflowtex}
% \mypopover{label}{note}: a note behind a button in the text. The button is
% a widget of kind popover; its label and the note are its parts, typeset by
% TeX out of the flow (examples/popover.js draws them). In print, a footnote.
\NewDocumentCommand\mypopover{m +m}{%
  \ifreflowtex
    \webwidget{popover}\webpart{label}{#1}\webpart{note}{\footnotesize #2}%
  \else
    \footnote{#2}%
  \fi}
