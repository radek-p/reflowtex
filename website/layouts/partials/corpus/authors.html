{{- /* SPDX-License-Identifier: AGPL-3.0-or-later */ -}} {{- /* Authors as a line: "A, B and C"; from five on, the first three and how many others. */ -}} {{- $a := . | default slice -}} {{- $n := len $a -}} {{- $s := "" -}} {{- if gt $n 4 -}} {{- $s = printf "%s and %d others" (delimit (first 3 $a) ", ") (sub $n 3) -}} {{- else if gt $n 1 -}} {{- $s = printf "%s and %s" (delimit (first (sub $n 1) $a) ", ") (index $a (sub $n 1)) -}} {{- else if eq $n 1 -}} {{- $s = index $a 0 -}} {{- end -}} {{- return $s -}}
