% reflowtex extraction template.
%
% This is the wrapper reflowtex compiles every snippet inside. The build pipeline
% (src/encode/pipeline.py) substitutes two placeholders, written further down as
% "PREAMBLE-MARK" and "CONTENT-MARK" below (each is the word PREAMBLE or CONTENT
% wrapped in double percent signs):
%
%   - the PREAMBLE marker  → the caller's preamble (extra packages, macros, fonts)
%   - the CONTENT marker   → the snippet body, set between begin/end document
%
% (The markers are not spelled literally in this comment because the substitution
% is a plain text replace and would rewrite them here too.)
%
% Everything the pipeline itself depends on (the serializer, the TikZ picture
% capture, the shipout hook) is set up *here*, before the user preamble, so a
% preamble is free to load its own packages and TikZ libraries on top.

\documentclass{article}

% ── Font resolution: prefer TeX Live's own copies over OS-installed ones ──────
% luaotfload resolves a font *by name* against a database built from several
% locations, searched in `location_precedence` order — whose default is
% {"system", "texmf", "local"}. On macOS, "system" means ~/Library/Fonts and
% /Library/Fonts — regular Font Book directories any app (or a user, installing
% a font by hand) can add to, independently of TeX Live. If a font of the same
% name ends up there too, it can be a *different, non-identical* copy of what
% TeX Live ships under texmf-dist (different glyph-index ordering internally,
% despite matching filenames). Since reflowtex addresses glyphs by the index
% LuaTeX assigns while compiling but *serves the browser the texmf-dist file*
% (via kpsewhich, see fonts.py), a "system" font winning that lookup makes every
% glyph index wrong — wrong enough that normalise_glyph_addressing
% (transforms.py) treats it as a substituted glyph and PUA-rewrites it, and the
% served font (lacking the PUA codepoints, since they were computed against a
% different file) can't display it at all. Reordering precedence so texmf wins
% keeps compilation and serving on the same file. This can't be set via a
% config file shipped in the repo (luaotfload reads its config before this
% document exists), so it's forced here, before any font is loaded. Only an
% issue if a same-named font happens to exist outside texmf-dist too (nothing
% to do with a stock TeX Live install), but forcing texmf first costs nothing
% when there's no such collision.
\directlua{
  if fonts and fonts.names and fonts.names.set_location_precedence then
    fonts.names.set_location_precedence({'texmf', 'system', 'local'})
    fonts.names.flush_lookup_cache()
  end
}

% ── Fonts & math ─────────────────────────────────────────────────────────────
% Text is set in Latin Modern Roman via fontspec (OpenType, modern glyph
% addressing — see the location-precedence fix above). Math intentionally stays
% on *classic* Computer Modern (cmex/cmsy/cmmi, plus amssymb's msam/msbm) rather
% than routing it through an OpenType math font: those 8-bit fonts are exactly
% what reflowtex's legacy Type1→OTF conversion exists for (fonts.py's
% legacy_otf(), t1_convert.py) — every glyph gets PUA-remapped into a
% purpose-built cmap the served font actually has, instead of depending on an
% OpenType math font's own glyph-assembly mechanism matching between compile
% time and what gets served (examples/testmath renders this way and is the
% reference for how the legacy path should look). A preamble that wants an
% OpenType math font instead can \usepackage{unicode-math} and \setmathfont{}
% itself; the template no longer preloads either.
\usepackage{mathtools}
\usepackage{amssymb}
\usepackage{fontspec}
\setmainfont{Latin Modern Roman}

% ── reflowtex node-list serializer ─────────────────────────────────────────────
% Exports the finished node list as output.json (see src/extract/serializer.lua).
\directlua{dofile("serializer.lua")}

% ── TikZ picture capture ─────────────────────────────────────────────────────
% A tikzpicture is already an hbox with the dimensions TeX computed; only its
% contents are unusable to us (raw pdf_literal drawing operators). So each
% picture is externalised to its own PDF, which the pipeline converts to SVG —
% the box keeps TeX's metrics and gains a payload the browser can draw. Because
% the result is still just a box, a picture can sit anywhere a box can: inline,
% inside math, inside an align row.
%
% Loaded here, before the user preamble, so a preamble can \usetikzlibrary{…}
% and \pgfdeclarelayer on top of it.
\usepackage{tikz}
\usetikzlibrary{external}
\tikzexternalize[prefix=pics/]
% Linking the node back to its file needs a hook, because the image node cannot
% name its own file (LuaTeX's img.fromnode is unavailable here). The hook must be
% \pgfincludeexternalgraphics: tikz's external library never calls
% \includegraphics, so patching that would silently do nothing.
\newcount\reflowtexPicCount
\let\reflowtexOrigIncludeExternal\pgfincludeexternalgraphics
\def\pgfincludeexternalgraphics#1{%
  \global\advance\reflowtexPicCount by 1
  \directlua{Serializer.note_picture(\the\reflowtexPicCount, "\luaescapestring{#1}")}%
  % The attribute rides on every node created inside the group, which is how the
  % image rule node is later matched to its file.
  \begingroup\attribute902=\reflowtexPicCount\relax\reflowtexOrigIncludeExternal{#1}\endgroup}

% The serializer needs the finished page to learn the document order of
% paragraphs and displays (displays never reach pre_linebreak_filter).
% \ShipoutBox is \chardef'd by the kernel, so \numexpr yields its number.
\AddToHook{shipout/before}{\directlua{Serializer.shipout(\the\numexpr\ShipoutBox\relax)}}

% ── User preamble ────────────────────────────────────────────────────────────
%%PREAMBLE%%

\begin{document}
%%CONTENT%%
\end{document}
