\begin{leantheorem}[decl=sum_odd, url=https://example.org/sum_odd]
\begin{theorem}
The sum of the first $n$ odd numbers is $n^2$.
\end{theorem}
\begin{proof}
By induction on $n$. For $n = 0$ both sides are $0$. If the sum of the first
$k$ odd numbers is $k^2$, adding the next one gives $k^2 + 2k + 1 = (k+1)^2$.
\end{proof}
\begin{leancode}
theorem sum_odd (n : ℕ) :
    ∑ i ∈ Finset.range n, (2 * i + 1) = n ^ 2 := by
  induction n with
  | zero => simp
  | succ k ih => rw [Finset.sum_range_succ, ih]; ring -- the last term
\end{leancode}
\end{leantheorem}
