% Accessible formulas: tagging on and luamml loaded, so that LaTeX writes
% every formula's MathML (with classic fonts LaTeX loads luamml only when
% asked). Reflow TeX reads it for screen readers.
\DocumentMetadata{tagging=on}
\tagpdfsetup{math/mathml/luamml/load=true}
\usepackage{amsthm}
\newtheorem{theorem}{Theorem}
\newtheorem*{claim}{Claim}
\usepackage[boxed]{reflowtex}
