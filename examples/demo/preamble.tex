% Reflow TeX demo preamble — prepended to every snippet in this directory.
%
% It shows that a preamble can load packages, TikZ libraries, and macros on top
% of the template (which loads tikz itself, before this preamble). \usetikzlibrary
% here is what makes that ordering matter.
\usepackage{xcolor}
\usepackage{tikz-cd}
\usetikzlibrary{arrows.meta}
\newcommand{\R}{\mathbb{R}}
\newcommand{\abs}[1]{\left\lvert #1 \right\rvert}
\newcommand{\norm}[1]{\left\lVert #1 \right\rVert}

% The companion package (src/latex/reflowtex.sty): web-first environments such
% as the collapsible proof in 07-collapsible-proof.tex. Every one of them falls
% back to its print form in an ordinary LaTeX run.
\usepackage{amsthm}
\newtheorem{theorem}{Theorem}
\usepackage{reflowtex}
