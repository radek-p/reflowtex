\begin{document}
\begin{tikzpicture}
\begin{axis}[width=5.6cm, height=3.6cm, xmin=-4, xmax=4, ymin=-0.2, ymax=6.6,
  axis lines=left, samples=120, xtick={-4,0,4}, ytick={0,6.28318},
  yticklabels={$0$,$2\pi$}, tick label style={font=\footnotesize},
  legend pos=north west, legend style={draw=none, fill=none, font=\footnotesize}]

\addlegendentry{$v=0$}

\addlegendentry{$v=0.8$}
\end{axis}
