\chapter{Counting}
\newcommand{\N}{\mathbb{N}}
We count with the natural numbers $\N = \{0, 1, 2, \dots\}$, and prove
things about them by induction.
\begin{theorem}\label{thm:induction}
If a property holds for $0$, and holds for $n+1$ whenever it holds for $n$,
then it holds for every $n \in \N$.
\end{theorem}
\begin{equation}\label{eq:gauss}
  0 + 1 + \dots + n = \frac{n(n+1)}{2}.
\end{equation}
